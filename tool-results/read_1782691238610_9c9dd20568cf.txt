     1→import React, { useState, useEffect, useRef } from "react";
     2→import * as pdfjsLib from "pdfjs-dist";
     3→import { 
     4→  Sparkles, PenTool, Type, Search, Eye, ShieldCheck, 
     5→  Trash2, Plus, Info, Check, RefreshCw, Type as TextIcon,
     6→  Layers, Lock, Compass, HelpCircle
     7→} from "lucide-react";
     8→
     9→interface InteractiveCanvasProps {
    10→  pdfBytes: Uint8Array;
    11→  totalPages: number;
    12→  onApplyTextAnnotation: (
    13→    pageNum: number,
    14→    text: string,
    15→    x: number,
    16→    y: number,
    17→    fontSize: number,
    18→    color: string,
    19→    useBg: boolean,
    20→    bgColor: string,
    21→    fontFamily?: string,
    22→    strokeColor?: string,
    23→    useStroke?: boolean
    24→  ) => Promise<void>;
    25→  onApplySignature: (
    26→    pageNum: number,
    27→    sigPngBytes: Uint8Array,
    28→    x: number,
    29→    y: number,
    30→    width: number,
    31→    height: number
    32→  ) => Promise<void>;
    33→  onApplyRedaction: (
    34→    words: string[],
    35→    boxColor: string,
    36→    opacity: number,
    37→    replaceText: string,
    38→    replaceTextColor: string,
    39→    replaceTextSize: string
    40→  ) => Promise<void>;
    41→  onApplyWatermark: (
    42→    text: string,
    43→    size: number,
    44→    color: string,
    45→    opacity: number
    46→  ) => Promise<void>;
    47→  onApplyPageNumbers: (
    48→    pos: string,
    49→    start: number,
    50→    size: number,
    51→    color: string
    52→  ) => Promise<void>;
    53→  onApplySearchReplace: (
    54→    search: string,
    55→    replace: string
    56→  ) => Promise<void>;
    57→  isProcessing: boolean;
    58→}
    59→
    60→export const InteractiveCanvas: React.FC<InteractiveCanvasProps> = ({
    61→  pdfBytes,
    62→  totalPages,
    63→  onApplyTextAnnotation,
    64→  onApplySignature,
    65→  onApplyRedaction,
    66→  onApplyWatermark,
    67→  onApplyPageNumbers,
    68→  onApplySearchReplace,
    69→  isProcessing
    70→}) => {
    71→  // Page selector state
    72→  const [selectedPage, setSelectedPage] = useState(1);
    73→  const [pageLoading, setPageLoading] = useState(false);
    74→  const [activeEditorMode, setActiveEditorMode] = useState<"signature" | "text" | "redact" | "search_replace" | "watermark">("signature");
    75→
    76→  // Page Dimension caches for scaling coordinate translations
    77→  const [pdfPageWidth, setPdfPageWidth] = useState(595); // default A4 width
    78→  const [pdfPageHeight, setPdfPageHeight] = useState(842); // default A4 height
    79→  const [canvasRenderWidth, setCanvasRenderWidth] = useState(400);
    80→  const [canvasRenderHeight, setCanvasRenderHeight] = useState(600);
    81→
    82→  // References
    83→  const previewCanvasRef = useRef<HTMLCanvasElement>(null);
    84→  const sigCanvasRef = useRef<HTMLCanvasElement>(null);
    85→
    86→  // Signature states
    87→  const [sigColor, setSigColor] = useState("#000000");
    88→  const [sigPenSize, setSigPenSize] = useState(2.5);
    89→  const [isDrawingSig, setIsDrawingSig] = useState(false);
    90→  const [sigHasInk, setSigHasInk] = useState(false);
    91→
    92→  // Text inputs
    93→  const [textToInsert, setTextToInsert] = useState("");
    94→  const [textFontSize, setTextFontSize] = useState(14);
    95→  const [textHexColor, setTextHexColor] = useState("#ea4335");
    96→  const [textUseBg, setTextUseBg] = useState(false);
    97→  const [textBgColor, setTextBgColor] = useState("#ffffff");
    98→  const [textFontFamily, setTextFontFamily] = useState("Tajawal");
    99→  const [textUseStroke, setTextUseStroke] = useState(false);
   100→  const [textStrokeColor, setTextStrokeColor] = useState("#000000");
   101→
   102→  // Signature & Custom Image states
   103→  const [signatureType, setSignatureType] = useState<"draw" | "upload">("draw");
   104→  const [uploadedImageBytes, setUploadedImageBytes] = useState<Uint8Array | null>(null);
   105→  const [uploadedImagePreview, setUploadedImagePreview] = useState<string | null>(null);
   106→  const [uploadedImageWidth, setUploadedImageWidth] = useState(120);
   107→  const [uploadedImageHeight, setUploadedImageHeight] = useState(120);
   108→
   109→  const handleImageUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
   110→    if (e.target.files && e.target.files[0]) {
   111→      const file = e.target.files[0];
   112→      const reader = new FileReader();
   113→      reader.onload = (event) => {
   114→        const dataUrl = event.target?.result as string;
   115→        setUploadedImagePreview(dataUrl);
   116→
   117→        // Load into an HTML Image to preserve aspect ratio and serialize as PNG bytes
   118→        const img = new Image();
   119→        img.onload = () => {
   120→          const canvas = document.createElement("canvas");
   121→          canvas.width = img.naturalWidth;
   122→          canvas.height = img.naturalHeight;
   123→          const ctx = canvas.getContext("2d");
   124→          if (ctx) {
   125→            ctx.drawImage(img, 0, 0);
   126→            const pngUrl = canvas.toDataURL("image/png");
   127→            const base64 = pngUrl.split(",")[1];
   128→            const binStr = atob(base64);
   129→            const bytes = new Uint8Array(binStr.length);
   130→            for (let i = 0; i < binStr.length; i++) {
   131→              bytes[i] = binStr.charCodeAt(i);
   132→            }
   133→            setUploadedImageBytes(bytes);
   134→            
   135→            const aspect = img.naturalHeight / img.naturalWidth;
   136→            setUploadedImageWidth(140);
   137→            setUploadedImageHeight(Math.round(140 * aspect));
   138→          }
   139→        };
   140→        img.src = dataUrl;
   141→      };
   142→      reader.readAsDataURL(file);
   143→    }
   144→  };
   145→
   146→  // Redaction inputs
   147→  const [redactWords, setRedactWords] = useState("");
   148→  const [redactBoxColor, setRedactBoxColor] = useState("#202124");
   149→  const [redactOpacity, setRedactOpacity] = useState(100);
   150→  const [redactUseReplace, setRedactUseReplace] = useState(false);
   151→  const [redactReplaceText, setRedactReplaceText] = useState("");
   152→  const [redactReplaceTextColor, setRedactReplaceTextColor] = useState("#ffffff");
   153→  const [redactReplaceTextSize, setRedactReplaceTextSize] = useState("auto");
   154→
   155→  // Search & Replace inputs
   156→  const [searchText, setSearchText] = useState("");
   157→  const [replaceText, setReplaceText] = useState("");
   158→
   159→  // Watermark inputs
   160→  const [wmText, setWmText] = useState("");
   161→  const [wmSize, setWmSize] = useState(48);
   162→  const [wmColor, setWmColor] = useState("#ea4335");
   163→  const [wmOpacity, setWmOpacity] = useState(25);
   164→
   165→  // Page numbering inputs
   166→  const [numPosition, setNumPosition] = useState("bottom-center");
   167→  const [numStart, setNumStart] = useState(1);
   168→  const [numSize, setNumSize] = useState(11);
   169→  const [numColor, setNumColor] = useState("#202124");
   170→
   171→  // --- RENDER SELECTED PAGE ON PREVIEW CANVAS ---
   172→  useEffect(() => {
   173→    let active = true;
   174→    setPageLoading(true);
   175→
   176→    const renderPage = async () => {
   177→      try {
   178→        const version = pdfjsLib.version || "4.0.379";
   179→        pdfjsLib.GlobalWorkerOptions.workerSrc = `https://unpkg.com/pdfjs-dist@${version}/build/pdf.worker.min.js`;
   180→
   181→        const loadingTask = pdfjsLib.getDocument({ data: pdfBytes });
   182→        const pdf = await loadingTask.promise;
   183→
   184→        if (!active) return;
   185→        const page = await pdf.getPage(selectedPage);
   186→
   187→        if (!active) return;
   188→        const canvas = previewCanvasRef.current;
   189→        if (!canvas) return;
   190→
   191→        const ctx = canvas.getContext("2d");
   192→        if (!ctx) return;
   193→
   194→        // Save native PDF point dimensions
   195→        const rawViewport = page.getViewport({ scale: 1.0 });
   196→        setPdfPageWidth(rawViewport.width);
   197→        setPdfPageHeight(rawViewport.height);
   198→
   199→        // Render at suitable viewing size (max 520px wide or max 650px high)
   200→        let scale = 520 / rawViewport.width;
   201→        if (rawViewport.height * scale > 650) {
   202→          scale = 650 / rawViewport.height;
   203→        }
   204→
   205→        const viewport = page.getViewport({ scale });
   206→        setCanvasRenderWidth(viewport.width);
   207→        setCanvasRenderHeight(viewport.height);
   208→
   209→        canvas.width = viewport.width;
   210→        canvas.height = viewport.height;
   211→
   212→        ctx.fillStyle = "#ffffff";
   213→        ctx.fillRect(0, 0, canvas.width, canvas.height);
   214→
   215→        const renderContext = {
   216→          canvasContext: ctx,
   217→          viewport: viewport,
   218→          canvas: canvas,
   219→        };
   220→
   221→        await page.render(renderContext).promise;
   222→        if (active) setPageLoading(false);
   223→      } catch (err) {
   224→        console.error("Failed to render page:", err);
   225→        if (active) setPageLoading(false);
   226→      }
   227→    };
   228→
   229→    renderPage();
   230→
   231→    return () => {
   232→      active = false;
   233→    };
   234→  }, [pdfBytes, selectedPage]);
   235→
   236→  // --- SIGNATURE DRAWING CANVAS LOGIC ---
   237→  useEffect(() => {
   238→    const canvas = sigCanvasRef.current;
   239→    if (!canvas) return;
   240→
   241→    const ctx = canvas.getContext("2d");
   242→    if (!ctx) return;
   243→
   244→    // Reset with grid background
   245→    resetSigCanvas();
   246→  }, [activeEditorMode]);
   247→
   248→  const resetSigCanvas = () => {
   249→    const canvas = sigCanvasRef.current;
   250→    if (!canvas) return;
   251→    const ctx = canvas.getContext("2d");
   252→    if (!ctx) return;
   253→
   254→    ctx.fillStyle = "#ffffff";
   255→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   256→
   257→    // Light guidelines
   258→    ctx.strokeStyle = "#f1f3f4";
   259→    ctx.lineWidth = 1;
   260→    for (let x = 0; x < canvas.width; x += 40) {
   261→      ctx.beginPath();
   262→      ctx.moveTo(x, 0);
   263→      ctx.lineTo(x, canvas.height);
   264→      ctx.stroke();
   265→    }
   266→    for (let y = 0; y < canvas.height; y += 40) {
   267→      ctx.beginPath();
   268→      ctx.moveTo(0, y);
   269→      ctx.lineTo(canvas.width, y);
   270→      ctx.stroke();
   271→    }
   272→
   273→    // Baseline indicator
   274→    ctx.strokeStyle = "#dadce0";
   275→    ctx.lineWidth = 1.5;
   276→    ctx.setLineDash([4, 4]);
   277→    ctx.beginPath();
   278→    ctx.moveTo(20, canvas.height - 35);
   279→    ctx.lineTo(canvas.width - 20, canvas.height - 35);
   280→    ctx.stroke();
   281→    ctx.setLineDash([]);
   282→
   283→    setSigHasInk(false);
   284→  };
   285→
   286→  const getEventCoords = (e: React.MouseEvent | React.TouchEvent | MouseEvent | TouchEvent, canvas: HTMLCanvasElement) => {
   287→    const rect = canvas.getBoundingClientRect();
   288→    let clientX, clientY;
   289→    if ("touches" in e) {
   290→      clientX = e.touches[0].clientX;
   291→      clientY = e.touches[0].clientY;
   292→    } else {
   293→      clientX = e.clientX;
   294→      clientY = e.clientY;
   295→    }
   296→    return {
   297→      x: (clientX - rect.left) * (canvas.width / rect.width),
   298→      y: (clientY - rect.top) * (canvas.height / rect.height)
   299→    };
   300→  };
   301→
   302→  const startSigDrawing = (e: React.MouseEvent<HTMLCanvasElement> | React.TouchEvent<HTMLCanvasElement>) => {
   303→    e.preventDefault();
   304→    setIsDrawingSig(true);
   305→    const canvas = sigCanvasRef.current;
   306→    if (!canvas) return;
   307→    const ctx = canvas.getContext("2d");
   308→    if (!ctx) return;
   309→
   310→    const { x, y } = getEventCoords(e.nativeEvent, canvas);
   311→    ctx.beginPath();
   312→    ctx.moveTo(x, y);
   313→    ctx.strokeStyle = sigColor;
   314→    ctx.lineWidth = sigPenSize;
   315→    ctx.lineCap = "round";
   316→    ctx.lineJoin = "round";
   317→    setSigHasInk(true);
   318→  };
   319→
   320→  const drawSig = (e: React.MouseEvent<HTMLCanvasElement> | React.TouchEvent<HTMLCanvasElement>) => {
   321→    if (!isDrawingSig) return;
   322→    e.preventDefault();
   323→    const canvas = sigCanvasRef.current;
   324→    if (!canvas) return;
   325→    const ctx = canvas.getContext("2d");
   326→    if (!ctx) return;
   327→
   328→    const { x, y } = getEventCoords(e.nativeEvent, canvas);
   329→    ctx.lineTo(x, y);
   330→    ctx.stroke();
   331→  };
   332→
   333→  const stopSigDrawing = () => {
   334→    setIsDrawingSig(false);
   335→  };
   336→
   337→  // --- CLICK PAGE PREVIEW TO STAMP/ANNOTATE ---
   338→  const handlePageClick = async (e: React.MouseEvent<HTMLCanvasElement>) => {
   339→    if (isProcessing) return;
   340→    const canvas = previewCanvasRef.current;
   341→    if (!canvas) return;
   342→
   343→    // Get click coords on rendering canvas scale
   344→    const rect = canvas.getBoundingClientRect();
   345→    const clickX = e.clientX - rect.left;
   346→    const clickY = e.clientY - rect.top;
   347→
   348→    // Translate coordinates to standard PDF coordinate space (BOTTOM-LEFT origin!)
   349→    // PDF width: pdfPageWidth, PDF height: pdfPageHeight
   350→    // Canvas render width: canvasRenderWidth, height: canvasRenderHeight
   351→    const scaleX = pdfPageWidth / canvasRenderWidth;
   352→    const scaleY = pdfPageHeight / canvasRenderHeight;
   353→
   354→    const pdfX = clickX * scaleX;
   355→    // In PDF space, Y=0 is at the BOTTOM of the page
   356→    const pdfY = (canvasRenderHeight - clickY) * scaleY;
   357→
   358→    if (activeEditorMode === "text") {
   359→      if (!textToInsert.trim()) {
   360→        alert("⚠️ يرجى كتابة نص التعليق أولاً في الحقل الجانبي!");
   361→        return;
   362→      }
   363→      setPageLoading(true);
   364→      await onApplyTextAnnotation(
   365→        selectedPage,
   366→        textToInsert,
   367→        pdfX,
   368→        pdfY,
   369→        textFontSize,
   370→        textHexColor,
   371→        textUseBg,
   372→        textBgColor,
   373→        textFontFamily,
   374→        textStrokeColor,
   375→        textUseStroke
   376→      );
   377→      setTextToInsert(""); // Clear text input after insertion
   378→    } else if (activeEditorMode === "signature") {
   379→      if (signatureType === "upload") {
   380→        if (!uploadedImageBytes || !uploadedImagePreview) {
   381→          alert("⚠️ يرجى تحميل صورة أو ختم أولاً من اللوحة الجانبية!");
   382→          return;
   383→        }
   384→        setPageLoading(true);
   385→        await onApplySignature(
   386→          selectedPage,
   387→          uploadedImageBytes,
   388→          pdfX - uploadedImageWidth / 2,
   389→          pdfY - uploadedImageHeight / 2,
   390→          uploadedImageWidth,
   391→          uploadedImageHeight
   392→        );
   393→      } else {
   394→        const sigCanvas = sigCanvasRef.current;
   395→        if (!sigCanvas || !sigHasInk) {
   396→          alert("⚠️ يرجى رسم توقيعك على اللوحة الجانبية أولاً قبل لصقه!");
   397→          return;
   398→        }
   399→        
   400→        setPageLoading(true);
   401→
   402→        // Convert drawing to raw PNG bytes
   403→        const dataUrl = sigCanvas.toDataURL("image/png");
   404→        const base64 = dataUrl.split(",")[1];
   405→        const binStr = atob(base64);
   406→        const pngBytes = new Uint8Array(binStr.length);
   407→        for (let i = 0; i < binStr.length; i++) {
   408→          pngBytes[i] = binStr.charCodeAt(i);
   409→        }
   410→
   411→        // Stamp with size around width=140pt
   412→        const sigW = 140;
   413→        const aspect = sigCanvas.height / sigCanvas.width;
   414→        const sigH = sigW * aspect;
   415→
   416→        // Adjust Y so the signature centers around the clicked point
   417→        await onApplySignature(
   418→          selectedPage,
   419→          pngBytes,
   420→          pdfX - sigW / 2,
   421→          pdfY - sigH / 2,
   422→          sigW,
   423→          sigH
   424→        );
   425→      }
   426→    }
   427→  };
   428→
   429→  // Redact Action
   430→  const triggerRedact = async () => {
   431→    if (!redactWords.trim()) {
   432→      alert("⚠️ يرجى إدخال كلمات للطمس أولاً!");
   433→      return;
   434→    }
   435→    const list = redactWords.split(",").map(w => w.trim()).filter(Boolean);
   436→    setPageLoading(true);
   437→    await onApplyRedaction(
   438→      list,
   439→      redactBoxColor,
   440→      redactOpacity,
   441→      redactUseReplace ? redactReplaceText : "",
   442→      redactReplaceTextColor,
   443→      redactReplaceTextSize
   444→    );
   445→    setRedactWords("");
   446→  };
   447→
   448→  // Search & Replace Action
   449→  const triggerSearchReplace = async () => {
   450→    if (!searchText.trim()) {
   451→      alert("⚠️ أدخل الكلمة للبحث عنها أولاً!");
   452→      return;
   453→    }
   454→    setPageLoading(true);
   455→    await onApplySearchReplace(searchText.trim(), replaceText.trim());
   456→    setSearchText("");
   457→    setReplaceText("");
   458→  };
   459→
   460→  // Watermark Action
   461→  const triggerWatermark = async () => {
   462→    if (!wmText.trim()) {
   463→      alert("⚠️ أدخل نص العلامة المائية أولاً!");
   464→      return;
   465→    }
   466→    setPageLoading(true);
   467→    await onApplyWatermark(wmText.trim(), wmSize, wmColor, wmOpacity);
   468→    setWmText("");
   469→  };
   470→
   471→  // Page Numbers Action
   472→  const triggerPageNumbers = async () => {
   473→    setPageLoading(true);
   474→    await onApplyPageNumbers(numPosition, numStart, numSize, numColor);
   475→  };
   476→
   477→  return (
   478→    <div className="grid grid-cols-1 lg:grid-cols-12 gap-6 items-stretch select-none">
   479→      
   480→      {/* 1. LEFT SIDE: THE INTERACTIVE DOCUMENT PREVIEW STAGE (8 COLS) */}
   481→      <div className="lg:col-span-7 flex flex-col items-center bg-gray-100 rounded-2xl border border-gray-200 p-4 min-h-[500px] justify-between shadow-inner">
   482→        
   483→        {/* Page selector bar */}
   484→        <div className="flex items-center gap-3 w-full justify-between border-b border-gray-200 pb-3 mb-3">
   485→          <div className="flex items-center gap-2">
   486→            <span className="flex h-6 w-6 items-center justify-center rounded-lg bg-google-blue text-white text-xs font-bold">
   487→              {selectedPage}
   488→            </span>
   489→            <span className="text-xs font-bold text-gray-700">معاينة وتعديل الصفحة الحالية</span>
   490→          </div>
   491→
   492→          <div className="flex items-center gap-2">
   493→            <button
   494→              onClick={() => setSelectedPage(p => Math.max(1, p - 1))}
   495→              disabled={selectedPage === 1 || pageLoading}
   496→              className="px-3 py-1 text-xs font-bold rounded-lg bg-white border border-gray-200 hover:bg-gray-50 disabled:opacity-40 cursor-pointer"
   497→            >
   498→              السابق
   499→            </button>
   500→            <span className="text-xs font-semibold text-gray-500">
   501→              {selectedPage} / {totalPages}
   502→            </span>
   503→            <button
   504→              onClick={() => setSelectedPage(p => Math.min(totalPages, p + 1))}
   505→              disabled={selectedPage === totalPages || pageLoading}
   506→              className="px-3 py-1 text-xs font-bold rounded-lg bg-white border border-gray-200 hover:bg-gray-50 disabled:opacity-40 cursor-pointer"
   507→            >
   508→              التالي
   509→            </button>
   510→          </div>
   511→        </div>
   512→
   513→        {/* The interactive canvas preview box */}
   514→        <div className="relative border border-gray-200 bg-white shadow-md rounded-xl overflow-hidden max-w-full group">
   515→          {pageLoading && (
   516→            <div className="absolute inset-0 bg-white/80 backdrop-blur-xs flex flex-col items-center justify-center z-10">
   517→              <RefreshCw className="h-8 w-8 text-google-blue animate-spin mb-2" />
   518→              <span className="text-xs font-bold text-gray-700">جاري رسم وتحضير محتوى الصفحة...</span>
   519→            </div>
   520→          )}
   521→
   522→          <canvas
   523→            ref={previewCanvasRef}
   524→            onClick={handlePageClick}
   525→            className={`block max-w-full ${
   526→              isProcessing ? "cursor-wait" : 
   527→              activeEditorMode === "text" ? "cursor-text" : "cursor-crosshair"
   528→            }`}
   529→            title={activeEditorMode === "text" ? "انقر على أي مكان بالصفحة لوضع النص المكتوب" : "انقر لوضع التوقيع هنا"}
   530→          />
   531→
   532→          {/* Interactive cursor helper overlay */}
   533→          {!pageLoading && !isProcessing && (
   534→            <div className="absolute inset-0 pointer-events-none opacity-0 group-hover:opacity-100 transition-opacity flex items-center justify-center">
   535→              <div className="absolute inset-0 bg-blue-500/3 mix-blend-multiply" />
   536→            </div>
   537→          )}
   538→        </div>
   539→
   540→        {/* Action hints */}
   541→        <div className="w-full flex items-center gap-2 mt-4 bg-blue-50 border border-blue-100 p-2.5 rounded-xl text-[11px] text-google-blue leading-relaxed font-semibold">
   542→          <Info className="h-4 w-4 shrink-0" />
   543→          {activeEditorMode === "signature" && (
   544→            <span>💡 ارسم توقيعك أو حمّل صورتك/ختمك بالجانب، ثم **انقر بنقرة واحدة** على أي مكان في المعاينة أعلاه لتثبيته بدقة متناهية.</span>
   545→          )}
   546→          {activeEditorMode === "text" && (
   547→            <span>💡 اكتب التعليق المطلوب بالجانب، حدد اللون والحجم، ثم **انقر بنقرة واحدة** على الصفحة أعلاه لوضعه في ذلك الموضع فوراً.</span>
