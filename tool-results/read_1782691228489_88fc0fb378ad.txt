   811→  const handleApplySearchReplace = async (search: string, replace: string) => {
   812→    if (!pdfBytes || !totalPages) return;
   813→    setIsProcessing(true);
   814→    try {
   815→      const doc = await PDFDocument.load(pdfBytes, { ignoreEncryption: true });
   816→      const pages = doc.getPages();
   817→
   818→      const loadingTask = pdfjsLib.getDocument({ data: pdfBytes });
   819→      const pdf = await loadingTask.promise;
   820→      let totalHits = 0;
   821→
   822→      for (let pi = 0; pi < totalPages; pi++) {
   823→        const page = pages[pi];
   824→        const pjsPage = await pdf.getPage(pi + 1);
   825→        const textContent = await pjsPage.getTextContent();
   826→
   827→        for (const item of textContent.items as any[]) {
   828→          const itemText = item.str;
   829→          const regex = new RegExp(search.replace(/[.*+?^${}()|[\]\\]/g, "\\$&"), "gi");
   830→          
   831→          if (regex.test(itemText)) {
   832→            const x = item.transform[4];
   833→            const y = item.transform[5];
   834→            const detectedSize = Math.abs(item.transform[3]) || 12;
   835→
   836→            const boxW = Math.max(item.width || (search.length * (detectedSize * 0.55)), 25);
   837→            const boxH = detectedSize + 4;
   838→
   839→            // Draw clean background to mask the text
   840→            page.drawRectangle({
   841→              x: x,
   842→              y: y - 2,
   843→              width: boxW,
   844→              height: boxH,
   845→              color: rgb(1, 1, 1), // White solid box
   846→            });
   847→
   848→            // Draw replacement text safely using Canvas-PNG engine
   849→            if (replace) {
   850→              const textPngUrl = drawTextAsPng(replace, detectedSize * 0.9, "#1a73e8", false, "");
   851→              if (textPngUrl) {
   852→                const binStr = atob(textPngUrl.split(",")[1]);
   853→                const textBytes = new Uint8Array(binStr.length);
   854→                for (let i = 0; i < binStr.length; i++) textBytes[i] = binStr.charCodeAt(i);
   855→
   856→                const textImg = await doc.embedPng(textBytes);
   857→                const scale = 0.4;
   858→                const textW = textImg.width * scale;
   859→                const textH = textImg.height * scale;
   860→
   861→                page.drawImage(textImg, {
   862→                  x: x + (boxW - textW) / 2,
   863→                  y: y + (boxH - textH) / 2 - 1,
   864→                  width: textW,
   865→                  height: textH,
   866→                });
   867→              }
   868→            }
   869→            totalHits++;
   870→          }
   871→        }
   872→      }
   873→
   874→      if (totalHits === 0) {
   875→        alert("⚠️ لم يتم العثور على أي مطابقة للكلمة للبحث عنها واستبدالها.");
   876→      } else {
   877→        const newBytes = await doc.save();
   878→        setPdfBytes(newBytes);
   879→        setFileSize(newBytes.byteLength);
   880→        alert(`✅ تم استبدال الكلمة بنجاح في ${totalHits} مواضع بالمستند!`);
   881→      }
   882→    } catch (err: any) {
   883→      alert("❌ فشل استبدال النص: " + err.message);
   884→    } finally {
   885→      setIsProcessing(false);
   886→    }
