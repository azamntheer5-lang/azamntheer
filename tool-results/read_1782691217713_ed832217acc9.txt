    41→    41→// Helper: Safely draw Arabic or Unicode text on off-screen HTML canvas
    42→    42→// to prevent "WinAnsiEncoding" crash in PDF-lib when writing non-ASCII characters.
    43→    43→const drawTextAsPng = (
    44→    44→  text: string, 
    45→    45→  fontSize: number, 
    46→    46→  colorHex: string, 
    47→    47→  useBg: boolean, 
    48→    48→  bgColorHex: string,
    49→    49→  fontFamily: string = "Tajawal",
    50→    50→  strokeColorHex: string = "",
    51→    51→  useStroke: boolean = false
    52→    52→): string | null => {
    53→    53→  const canvas = document.createElement("canvas");
    54→    54→  const ctx = canvas.getContext("2d");
    55→    55→  if (!ctx) return null;
    56→    56→
    57→    57→  // Scale up rendering for crisp visual fidelity in high resolution
    58→    58→  const renderScale = 2.8;
    59→    59→  const scaledSize = fontSize * renderScale;
    60→    60→
    61→    61→  ctx.font = `bold ${scaledSize}px "${fontFamily}", "Cairo", "Tajawal", sans-serif`;
    62→    62→  const metrics = ctx.measureText(text);
    63→    63→  const textWidth = metrics.width;
    64→    64→  // Account for ascents/descents elegantly
    65→    65→  const textHeight = scaledSize * 1.5;
    66→    66→
    67→    67→  canvas.width = textWidth + (24 * renderScale);
    68→    68→  canvas.height = textHeight + (10 * renderScale);
    69→    69→
    70→    70→  // Clear and paint optional background
    71→    71→  if (useBg) {
    72→    72→    ctx.fillStyle = bgColorHex;
    73→    73→    ctx.fillRect(0, 0, canvas.width, canvas.height);
    74→    74→  } else {
    75→    75→    ctx.clearRect(0, 0, canvas.width, canvas.height);
    76→    76→  }
    77→    77→
    78→    78→  // Draw optional outline border
    79→    79→  if (useStroke && strokeColorHex) {
    80→    80→    ctx.strokeStyle = strokeColorHex;
    81→    81→    ctx.lineWidth = 1.8 * renderScale;
    82→    82→    ctx.strokeRect(
    83→    83→      0.9 * renderScale, 
    84→    84→      0.9 * renderScale, 
    85→    85→      canvas.width - 1.8 * renderScale, 
    86→    86→      canvas.height - 1.8 * renderScale
    87→    87→    );
    88→    88→  }
    89→    89→
    90→    90→  // Redraw configurations as resizing canvas resets state
    91→    91→  ctx.font = `bold ${scaledSize}px "${fontFamily}", "Cairo", "Tajawal", sans-serif`;
    92→    92→  ctx.fillStyle = colorHex;
    93→    93→  ctx.textBaseline = "middle";
    94→    94→  ctx.textAlign = "center";
    95→    95→  
    96→    96→  ctx.fillText(text, canvas.width / 2, canvas.height / 2);
    97→    97→
    98→    98→  return canvas.toDataURL("image/png");
    99→    99→};

   101→   101→// Helper: Extract all PDF text in the background for Gemini
   102→   102→const extractFullText = async (bytes: Uint8Array): Promise<string> => {
   103→   103→  try {
   104→   104→    const version = pdfjsLib.version || "4.0.379";
   105→   105→    pdfjsLib.GlobalWorkerOptions.workerSrc = `https://unpkg.com/pdfjs-dist@${version}/build/pdf.worker.min.js`;
   106→   106→
   107→   107→    const loadingTask = pdfjsLib.getDocument({ data: bytes });
   108→   108→    const pdf = await loadingTask.promise;
   109→   109→    let text = "";
   110→   110→    
   111→   111→    // Scan maximum of first 20 pages to keep processing swift & within token ranges
   112→   112→    const pagesToScan = Math.min(pdf.numPages, 20);
   113→   113→    for (let i = 1; i <= pagesToScan; i++) {
   114→   114→      const page = await pdf.getPage(i);
   115→   115→      const content = await page.getTextContent();
   116→   116→      const pageText = content.items.map((it: any) => it.str).join(" ");
   117→   117→      text += `\n[Page ${i}]\n${pageText}\n`;
   118→   118→    }
   119→   119→    return text;
   120→   120→  } catch (err) {
   121→   121→    console.error("Background text extraction failed:", err);
   122→   122→    return "";
   123→   123→  }
